#### 1.3 Brain Affinity Suggestion

**File:** `core/learning/plugin_suggestions.py` (NEW)

```python
from typing import List, Set
from core.learning.operator_fingerprint import OperatorFingerprint

def suggest_plugins(
    fingerprint: OperatorFingerprint,
    registry: PluginRegistry,
    top_k: int = 5,
) -> List[str]:
    """
    Suggest plugins based on operator's task affinities.
    
    Algorithm:
    1. For each task the operator is strong in, find plugins for that domain
    2. Score plugins by relevance + rating
    3. Return top-K non-installed plugins
    """
    
    if not fingerprint or fingerprint.sample_count < 50:
        return []  # not enough fingerprint data
    
    # Extract strong task types
    strong_tasks = [
        task_type
        for task_type, affinity in fingerprint.task_affinities.items()
        if affinity.strength_tier == "strong"
        and affinity.confidence > 0.7
    ]
    
    if not strong_tasks:
        return []
    
    # Map task types to plugin categories
    category_scores: Dict[str, float] = {}
    for task in strong_tasks:
        category = TASK_TYPE_TO_CATEGORY[task]
        category_scores[category] = category_scores.get(category, 0) + 1
    
    # Get top 2 categories by frequency
    top_categories = sorted(
        category_scores.items(),
        key=lambda x: x[1],
        reverse=True,
    )[:2]
    
    top_category_names = [cat for cat, _ in top_categories]
    
    # Find plugins in top categories
    plugins = registry.list_plugins_by_category(top_category_names)
    
    # Score by rating
    plugins_by_score = sorted(
        plugins,
        key=lambda p: (p.rating_average, p.download_count),
        reverse=True,
    )[:top_k]
    
    return [p.plugin_id for p in plugins_by_score]
```

```python
from dataclasses import dataclass
from typing import List, Dict
import json

@dataclass(frozen=True)
class SeccompProfile:
    """Immutable seccomp profile for a plugin."""
    
    plugin_id: str
    allowed_syscalls: List[str]
    denied_syscalls: List[str]
    filesystem_rules: Dict[str, str]    # path -> "r" | "rw"
    cpu_limit_percent: int              # [1..100]
    memory_limit_mb: int                # [64..512]
    timeout_seconds: int
    
    def to_bpf_rules(self) -> str:
        """
        Translate to BPF bytecode for seccomp filter.
        
        Format:
        ```
        # arch x86_64
        # action SCMP_ACT_KILL_THREAD
        
        read: allow
        write: allow
        open: allow
        close: allow
        ...
        module_load: kill
        admin_request: kill
        ```
        """
        pass  # BPF compilation deferred to Phase 2.2

def generate_profile_from_metadata(
    metadata: PluginMetadata,
) -> SeccompProfile:
    """
    Infer seccomp profile from plugin's declared needs.
    
    Algorithm:
    1. Start with a base allow-list (open, read, write, close, mmap, exit)
    2. Add syscalls from plugin's required_syscalls
    3. Add filesystem-related syscalls (stat, lstat, access)
    4. Hard deny: module_load, admin ops, network admin
    5. Clamp resource limits to reasonable defaults
    """
    
    base_syscalls = {
        "read", "write", "open", "close", "mmap", "munmap",
        "exit_group", "rt_sigaction", "rt_sigprocmask",
        "brk", "ioctl", "pread64", "pwrite64",
    }
    
    allowed = base_syscalls.copy()
    allowed.update(metadata.required_syscalls)
    
    # Add filesystem syscalls if needed
    if metadata.filesystem_paths:
        allowed.update({
            "stat", "lstat", "access", "faccessat",
            "openat", "renameat", "unlinkat", "mkdirat",
        })
    
    # Hard deny list
    denied = {
        "init_module", "delete_module", "finit_module",
        "kmod_load_module", "cap_drop", "setuid", "setgid",
        "socket", "connect", "bind", "listen",  # raw network
        "ioctl",  # privileged I/O
    }
    
    # If no network access declared, deny all network syscalls
    if not metadata.network_access:
        denied.update({
            "socket", "socketpair", "connect", "accept",
            "bind", "listen", "sendto", "recvfrom",
        })
    
    # Remove hard-deny from allowed
    allowed -= denied
    
    return SeccompProfile(
        plugin_id=metadata.plugin_id,
        allowed_syscalls=sorted(list(allowed)),
        denied_syscalls=sorted(list(denied)),
        filesystem_rules=metadata.filesystem_paths or {},
        cpu_limit_percent=min(100, 20),  # default 20%
        memory_limit_mb=min(512, 256),   # default 256MB
        timeout_seconds=min(3600, 60),   # default 60s
    )
```

**File:** `core/plugins/sandbox/executor.py` (NEW)

```python
import subprocess
import signal
import resource
from typing import Optional, Dict, Any
import tempfile

class SandboxExecutor:
    """Execute plugin code in isolated seccomp sandbox."""
    
    async def execute(
        self,
        plugin_id: str,
        operation: str,
        args: Dict[str, Any],
        profile: SeccompProfile,
    ) -> Dict[str, Any]:
        """
        Execute plugin operation in sandbox.
        
        Returns:
            {
                "status": "success" | "timeout" | "error" | "killed",
                "result": {...},
                "stderr": "...",
                "exit_code": int,
                "execution_time_ms": float,
            }
        """
        
        # Step 1: Prepare sandbox environment
        with tempfile.TemporaryDirectory() as tmpdir:
            # Copy plugin code to tmpdir
            plugin_code_path = self._copy_plugin_to_sandbox(
                plugin_id,
                tmpdir,
            )
            
            # Write seccomp profile
            profile_path = os.path.join(tmpdir, "seccomp.json")
            with open(profile_path, "w") as f:
                json.dump({
                    "allowed": profile.allowed_syscalls,
                    "denied": profile.denied_syscalls,
                }, f)
            
            # Step 2: Build subprocess args
            env = {
                "PLUGIN_ID": plugin_id,
                "SANDBOX_HOME": tmpdir,
            }
            
            cmd = [
                "seccomp-daemon",
                "--profile", profile_path,
                "--cpu-limit", str(profile.cpu_limit_percent),
                "--mem-limit", str(profile.memory_limit_mb),
                "--timeout", str(profile.timeout_seconds),
                "--chroot", tmpdir,
                "python3", plugin_code_path,
                "--operation", operation,
                "--args", json.dumps(args),
            ]
            
            # Step 3: Execute with timeout
            start = time.time()
            try:
                result = await asyncio.wait_for(
                    asyncio.create_subprocess_exec(
                        *cmd,
                        stdout=asyncio.subprocess.PIPE,
                        stderr=asyncio.subprocess.PIPE,
                        env=env,
                    ),
                    timeout=profile.timeout_seconds + 2,  # grace period
                )
                
                stdout, stderr = await result.communicate()
                elapsed = time.time() - start
                
                if result.returncode == 137:  # killed by signal
                    return {
                        "status": "killed",
                        "stderr": stderr.decode(),
                        "execution_time_ms": elapsed * 1000,
                    }
                
                return {
                    "status": "success" if result.returncode == 0 else "error",
                    "result": json.loads(stdout.decode()),
                    "stderr": stderr.decode(),
                    "exit_code": result.returncode,
                    "execution_time_ms": elapsed * 1000,
                }
            
            except asyncio.TimeoutError:
                return {
                    "status": "timeout",
                    "stderr": f"Timeout after {profile.timeout_seconds}s",
                    "execution_time_ms": (time.time() - start) * 1000,
                }
```

```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any, Dict, Optional
from datetime import datetime

@dataclass(frozen=True)
class ExecutionContext:
    """Immutable context passed to every plugin operation."""
    
    operation_id: str
    plugin_id: str
    operator_id: str
    tenant_id: str
    
    # Plugin metadata
    version: str
    sandbox_profile: SeccompProfile
    
    # Execution metadata
    started_at: datetime
    deadline: datetime  # timeout
    
    # Operator state (read-only)
    fingerprint: Optional[OperatorFingerprint]
    installed_plugins: List[str]  # plugin_ids
    
    # Audit
    audit_hash: str

@dataclass(frozen=True)
class PluginResponse:
    """Immutable plugin operation result."""
    
    status: Literal["success", "error", "retry"]
    data: Optional[Dict[str, Any]]  # success result
    error: Optional[str]             # error message
    error_code: Optional[str]        # structured error code
    metadata: Dict[str, Any]         # execution stats
    audit_hash: str

class PluginBase(ABC):
    """Base class for all v2 plugins."""
    
    __version__ = "2"
    
    @abstractmethod
    async def init(self, context: ExecutionContext) -> None:
        """
        Initialize plugin for this execution.
        
        Called once per operation, before any handlers.
        Perform setup, load config, validate environment.
        """
        pass
    
    @abstractmethod
    async def on_operator_decision(
        self,
        context: ExecutionContext,
        decision_type: str,
        options: List[Dict[str, Any]],
        chosen: Dict[str, Any],
    ) -> PluginResponse:
        """
        Hook: operator made a decision.
        
        Called after decision audit recorded.
        Plugin may:
        - Log analytics
        - Update fingerprint locally
        - Suggest alternatives (veto not allowed)
        """
        pass
    
    async def on_turn_start(
        self,
        context: ExecutionContext,
        task_type: str,
    ) -> PluginResponse:
        """Hook: turn started."""
        return PluginResponse(
            status="success",
            data={},
            error=None,
            error_code=None,
            metadata={},
            audit_hash=context.audit_hash,
        )
    
    async def on_turn_end(
        self,
        context: ExecutionContext,
        success: bool,
        result: Dict[str, Any],
    ) -> PluginResponse:
        """Hook: turn ended."""
        return PluginResponse(
            status="success",
            data={},
            error=None,
            error_code=None,
            metadata={},
            audit_hash=context.audit_hash,
        )
    
    async def shutdown(self, context: ExecutionContext) -> None:
        """
        Shutdown plugin cleanly.
        
        Called once per operation, after all handlers.
        Perform cleanup, close connections, save state.
        """
        pass

class PluginRegistration:
    """Declarative plugin registration (v2)."""
    
    @staticmethod
    def register(
        plugin_id: str,
        version: str,
        category: str,
        description: str,
        author: str,
    ) -> type:
        """Decorator to register a plugin class."""
        
        def decorator(cls):
            # Validate interface
            assert hasattr(cls, "init")
            assert hasattr(cls, "on_operator_decision")
            assert hasattr(cls, "shutdown")
            
            # Store metadata
            cls._plugin_id = plugin_id
            cls._plugin_version = version
            cls._plugin_category = category
            cls._plugin_description = description
            cls._plugin_author = author
            
            return cls
        
        return decorator

# Example plugin (auth-saml v2)

@PluginRegistration.register(
    plugin_id="auth-saml",
    version="2.0.0",
    category="Authentication",
    description="SAML 2.0 authentication plugin",
    author="CoreAuthor",
)
class SamlAuthPlugin(PluginBase):
    """SAML 2.0 authentication."""
    
    async def init(self, context: ExecutionContext) -> None:
        self.config = await load_config("auth-saml.yaml")
        self.sp = await init_service_provider(self.config)
    
    async def on_operator_decision(
        self,

    ) -> PluginResponse:
        if decision_type == "auth_method":

                operator_id=context.operator_id,

            )
        
        return PluginResponse(
            status="success",
            data={},
            error=None,
            error_code=None,
            metadata={"auth_method": chosen.get("method")},
            audit_hash=context.audit_hash,
        )
    
    async def shutdown(self, context: ExecutionContext) -> None:
        await self.sp.close()
```

**File:** `core/plugins/api/deprecation.py` (NEW)

```python
@dataclass(frozen=True)
class DeprecationPolicy:
    """
    Plugin API deprecation: 2-version grace period.
    
    Example:
    - v0.7.0 ships Plugin API v2
    - v0.8.x: v1 plugins load with warnings, v2 is primary
    - v0.9.0: v1 plugins removed, v2 mandatory
    - v1.0.0+: v3 introduced (if ever), v2 enters grace period
    """
    
    current_version: int              # e.g., 2
    deprecated_versions: List[int]    # e.g., [1]
    removed_versions: List[int]       # e.g., []

async def validate_plugin_version(
    plugin: PluginMetadata,
    policy: DeprecationPolicy,
) -> Tuple[bool, Optional[str]]:
    """
    Check if plugin version is allowed.
    
    Returns:
        (is_allowed, warning_message)
    """
    
    plugin_api_version = plugin.api_version  # from metadata
    
    if plugin_api_version == policy.current_version:
        return (True, None)
    
    if plugin_api_version in policy.deprecated_versions:
        return (
            True,
            f"Plugin API v{plugin_api_version} is deprecated. "
            f"Migrate to v{policy.current_version} by v0.9.0.",
        )
    
    if plugin_api_version in policy.removed_versions:
        return (False, f"Plugin API v{plugin_api_version} no longer supported.")
    
    return (False, f"Unknown plugin API version {plugin_api_version}")
```

```python
@dataclass(frozen=True)
class PluginQualityScore:
    """Immutable plugin quality assessment."""
    
    plugin_id: str
    score: float                # [0.0..1.0]
    
    # Components
    api_correctness: float      # Does it implement v2 interface?
    security_sandbox: float     # Passes adversarial tests?
    performance: float          # <200ms overhead?
    reliability: float          # Crash rate <0.1%?
    documentation: float        # README, examples?
    
    assessed_at: datetime
    audit_hash: str

class PluginQualityGrader:
    """Automated quality assessment."""
    
    async def grade_plugin(
        self,
        plugin_id: str,
        profile: SeccompProfile,
    ) -> PluginQualityScore:
        """
        Grade plugin on 5 dimensions.
        
        Algorithm:
        1. API correctness: parse metadata, validate interface
        2. Security: run 100-scenario adversarial tests
        3. Performance: measure overhead (10 operations)
        4. Reliability: stress test (100 operations)
        5. Documentation: check for README, examples
        """
        
        api_score = await self._grade_api(plugin_id)
        security_score = await self._grade_security(plugin_id, profile)
        perf_score = await self._grade_performance(plugin_id)
        reliability_score = await self._grade_reliability(plugin_id)
        docs_score = await self._grade_documentation(plugin_id)
        
        final_score = (
            0.25 * api_score +
            0.30 * security_score +
            0.20 * perf_score +
            0.15 * reliability_score +
            0.10 * docs_score
        )
        
        return PluginQualityScore(
            plugin_id=plugin_id,
            score=min(1.0, max(0.0, final_score)),
            api_correctness=api_score,
            security_sandbox=security_score,
            performance=perf_score,
            reliability=reliability_score,
            documentation=docs_score,
            assessed_at=now(),
            audit_hash=await audit_hash_chain(),
        )
```

```python
    report: PluginFlagReport,

    Process a flag report.

        **dataclasses.asdict(report),
        status="investigating",
        updated_at=now(),
```

```python
from dataclasses import dataclass
from typing import Optional
from datetime import datetime, timedelta

@dataclass(frozen=True)
class PluginHealthMetrics:
    """Immutable plugin health snapshot."""
    
    plugin_id: str
    
    # Usage
    active_users: int               # operators with enabled installation
    total_installs: int
    uninstall_count: int
    
    # Quality
    crash_rate: float               # [0.0..1.0] (operations crashed)
    avg_operation_time_ms: float
    p99_operation_time_ms: float
    
    # Ratings
    avg_rating: float               # [1.0..5.0]
    rating_count: int
    
    # Flags
    active_flag_count: int
    
    # Revenue (if applicable)
    monthly_revenue_cents: int      # amount in cents
    
    measured_at: datetime
    audit_hash: str

class PluginHealthMonitor:
    """Track plugin health in real-time."""
    
    async def measure_plugin_health(
        self,
        plugin_id: str,
        window_hours: int = 24,
    ) -> PluginHealthMetrics:
        """
        Measure plugin health across all operators.
        
        Algorithm:
        1. Count active installations
        2. Compute crash rate from execution logs
        3. Measure operation latencies (p50, p95, p99)
        4. Fetch rating metrics from PluginReviewStore
        5. Count open flag reports
        """
        
        # Active users
        active_users = await PluginInstallationStore.count_active(
            plugin_id=plugin_id,
        )
        
        total_installs = await PluginInstallationStore.count_all(
            plugin_id=plugin_id,
        )
        
        # Crashes
        execution_logs = await ExecutionLogStore.query(
            plugin_id=plugin_id,
            hours=window_hours,
        )
        
        crashed = sum(
            1 for log in execution_logs
            if log.status == "error" or log.status == "killed"
        )
        
        crash_rate = crashed / len(execution_logs) if execution_logs else 0
        
        # Latencies
        latencies = [log.execution_time_ms for log in execution_logs]
        avg_latency = statistics.mean(latencies) if latencies else 0
        p99_latency = numpy.percentile(latencies, 99) if latencies else 0
        
        # Ratings
        reviews = await PluginReviewStore.list_reviews(plugin_id)
        avg_rating = (
            statistics.mean([r.rating for r in reviews])
            if reviews else 0
        )
        
        # Flags
        flags = await PluginFlagStore.list_flags(
            plugin_id=plugin_id,
            status__in=["reported", "investigating"],
        )
        
        return PluginHealthMetrics(
            plugin_id=plugin_id,
            active_users=active_users,
            total_installs=total_installs,
            uninstall_count=total_installs - active_users,
            crash_rate=crash_rate,
            avg_operation_time_ms=avg_latency,
            p99_operation_time_ms=p99_latency,
            avg_rating=avg_rating,
            rating_count=len(reviews),
            active_flag_count=len(flags),
            monthly_revenue_cents=await self._compute_revenue(plugin_id),
            measured_at=now(),
            audit_hash=await audit_hash_chain(),
        )
```

```python
@dataclass(frozen=True)

class AuthorPayout:
    """Monthly payout to plugin author."""
    
    payout_id: str
    author_id: str
    year_month: str
    
    total_payout_cents: int         # sum of all plugins
    status: Literal[
        "pending",
        "processing",
        "completed",
        "failed",
    ]

    transaction_id: Optional[str]
    
    created_at: datetime
    completed_at: Optional[datetime]
    audit_hash: str

async def process_monthly_payouts() -> None:
    """
    Process monthly payouts to all plugin authors.
    
    Runs on 1st of each month.
    """
    
    # Get all plugins with revenue
    plugins = await PluginRegistry.list_all_plugins()
    
    author_totals: Dict[str, int] = {}
    
    for plugin in plugins:
        revenue = await compute_plugin_revenue(
            plugin.plugin_id,
            year_month=last_month(),
        )
        
        author_id = plugin.author_id
        if author_id not in author_totals:
            author_totals[author_id] = 0
        author_totals[author_id] += revenue.author_payout_cents
    
    # Create payouts
    for author_id, payout_cents in author_totals.items():
        payout = AuthorPayout(
            payout_id=uuid4().hex,
            author_id=author_id,
            year_month=last_month(),
            total_payout_cents=payout_cents,
            status="pending",
            payment_method=await get_author_payment_method(author_id),
            transaction_id=None,
            created_at=now(),
            completed_at=None,
            audit_hash=await audit_hash_chain(),
        )
        
        await AuthorPayoutStore.save(payout)
        
        # Process payment (Stripe, PayPal, etc.)
        await process_payout(payout)
```
