## Executive Summary

v0.9 builds a **real-time dashboard** that gives operators live visibility into Brain subsystem health, decision streams, cost burn, and interrupt controls. The system streams events via WebSocket, calculates live metrics, and allows operators to pause/resume/redirect turns mid-execution.

**Key invariant:** Dashboard is observational (read-only) until interrupt endpoints are called. No data is modified by viewing.

```python
@dataclass(frozen=True)
class DecisionStreamEvent:
    """Immutable event for real-time dashboard."""
    
    event_id: str  # UUID
    event_type: Literal[
        "decision_made",      # Operator chose option
        "cost_incurred",      # Turn cost calculated
        "confidence_updated", # Model confidence changed
        "plugin_invoked",     # Plugin suggested
        "engine_switched",    # Delegation occurred
        "interrupt_action"    # Pause/resume/redirect
    ]
    
    # Context
    turn_id: str
    operator_id: str
    tenant_id: str
    timestamp: datetime
    sequence_num: int  # Monotonic, ensures order
    
    # Payload (type-specific)
    payload: Dict[str, Any]  # {'decision_id', 'confidence'} etc.
    
    # Audit
    audit_hash: str


class DecisionBus:
    """Publishes events, manages subscriber queue."""
    
    def __init__(self, max_queue_size: int = 10_000):
        self.queue: asyncio.Queue = asyncio.Queue(maxsize=max_queue_size)
        self.sequence_counter: int = 0
    
    async def publish(self, event: DecisionStreamEvent) -> None:
        """Emit event to all subscribers (fire-and-forget if queue full)."""
        self.sequence_counter += 1
        event_with_seq = dataclasses.replace(
            event,
            sequence_num=self.sequence_counter
        )
        
        try:
            self.queue.put_nowait(event_with_seq)
        except asyncio.QueueFull:
            # Queue full, drop oldest event (circular buffer)
            try:
                self.queue.get_nowait()
            except asyncio.QueueEmpty:
                pass
            self.queue.put_nowait(event_with_seq)
    
    async def subscribe(self, from_seq: int = -1) -> AsyncIterator[DecisionStreamEvent]:
        """Subscribe to event stream (resume from sequence N if provided)."""
        # Note: In production, implement persistent event log for resume
        local_queue = asyncio.Queue()
        async for event in self.queue:
            if event.sequence_num > from_seq:
                await local_queue.put(event)
                yield event


# Integration points:
# - `core/orchestration/execution_context.py`: Emit on decision
# - `core/cost/cost_controller.py`: Emit cost_incurred
# - `core/learning/learning_engine.py`: Emit confidence_updated
# - `core/plugins/plugin_scheduler.py`: Emit plugin_invoked
# - `core/delegation/delegation_policy.py`: Emit engine_switched
# - WebSocket handler: Subscribe to bus
```

#### 1.2 Health Monitor Subsystem

**File:** `core/health/health_monitor.py` (NEW)

```python
@dataclass(frozen=True)
class SubsystemHealth:
    """Health snapshot of one subsystem."""
    
    name: str  # "HealthMonitor", "ContextBridge", etc.
    status: Literal["GREEN", "YELLOW", "RED"]
    
    # Latency percentiles (milliseconds)
    latency_p50: float
    latency_p95: float
    latency_p99: float
    
    # Throughput & errors
    throughput_hz: float  # Operations per second
    error_rate: float     # [0.0..1.0], fraction of errors
    last_error: Optional[str]
    error_count: int
    
    # Status determined by:
    # GREEN:  p99 < 100ms AND error < 0.5% AND throughput > 100 Hz
    # YELLOW: p99 < 200ms AND error < 2.0% AND throughput > 50 Hz
    # RED:    p99 > 200ms OR error > 2.0% OR throughput < 50 Hz
    
    timestamp: datetime


class HealthMonitor(Subsystem):
    """Polls all subsystems, maintains health history."""
    
    def __init__(self, brain: Brain, history_size: int = 60):
        self.brain = brain
        self.history: Dict[str, Deque[SubsystemHealth]] = {}
        for subsys_name in self.brain.subsystems.keys():
            self.history[subsys_name] = deque(maxlen=history_size)
    
    async def collect_health(self) -> Dict[str, SubsystemHealth]:
        """Poll all subsystems, return current health snapshot."""
        health = {}
        for name, subsys in self.brain.subsystems.items():
            # Call subsystem's get_metrics() endpoint
            metrics = await subsys.get_metrics()
            
            # Calculate percentiles from circular buffer
            latencies = [m.latency_ms for m in metrics.recent_operations]
            latencies.sort()
            
            error_count = sum(1 for m in metrics.recent_operations if m.error)
            error_rate = error_count / len(metrics.recent_operations)
            
            # Determine status
            p99 = latencies[int(0.99 * len(latencies))]
            if p99 < 100 and error_rate < 0.005:
                status = "GREEN"
            elif p99 < 200 and error_rate < 0.02:
                status = "YELLOW"
            else:
                status = "RED"
            
            health[name] = SubsystemHealth(
                name=name,
                status=status,
                latency_p50=latencies[int(0.50 * len(latencies))],
                latency_p95=latencies[int(0.95 * len(latencies))],
                latency_p99=p99,
                throughput_hz=len(metrics.recent_operations) / metrics.collection_duration_sec,
                error_rate=error_rate,
                last_error=metrics.last_error,
                error_count=error_count,
                timestamp=datetime.now()
            )
            
            self.history[name].append(health[name])
        
        return health
    
    async def run_polling_loop(self):
        """Poll every 1 second, emit HealthUpdated events."""
        while True:
            health = await self.collect_health()
            # Emit to decision bus or other listeners
            await self.emit_health_update(health)
            await asyncio.sleep(1.0)  # 1 Hz polling
```

**Integration points:**
- Run in background: `core/orchestration/execution_context.py` starts polling loop
- Expose API: `GET /api/dashboard/health` returns latest snapshot
- Subscribe: WebSocket handler subscribes to health updates

```python
from fastapi import WebSocket

@app.websocket("/ws/decision-stream")
async def websocket_endpoint(websocket: WebSocket, 
                             operator_id: str,
                             from_seq: int = -1):
    """Stream decisions + metrics in real-time."""
    await websocket.accept()
    
    try:
        bus = get_decision_bus()
        
        # Subscribe to event stream
        async for event in bus.subscribe(from_seq=from_seq):
            # Filter by operator (privacy)
            if event.operator_id != operator_id:
                continue
            
            # Serialize and send
            message = json.dumps({
                'type': event.event_type,
                'event_id': event.event_id,
                'turn_id': event.turn_id,
                'sequence_num': event.sequence_num,
                'timestamp': event.timestamp.isoformat(),
                'payload': event.payload
            })
            
            await websocket.send_text(message)
            
    except WebSocketDisconnect:
        pass
    except Exception as e:
        await websocket.send_json({'error': str(e)})
        await websocket.close(code=1011)
```

```python
@app.get("/api/dashboard/cost")
async def get_cost_status(operator_id: str) -> CostStatus:
    """Return current cost status for dashboard."""
    cost_controller = get_cost_controller()
    
    budget = cost_controller.get_budget(operator_id)
    spent = cost_controller.get_spent(operator_id)
    
    # Burn rate ($ per hour)
    window = timedelta(hours=1)
    recent_costs = cost_controller.get_costs(operator_id, since=datetime.now() - window)
    burn_rate = sum(c.amount for c in recent_costs)
    
    # Projection
    if burn_rate > 0:
        hours_remaining = (budget - spent) / burn_rate
        projected_total = (burn_rate * 24) + spent
    else:
        hours_remaining = float('inf')
        projected_total = spent
    
    return CostStatus(
        budget_usd=budget,
        spent_usd=spent,
        spent_percent=spent / budget if budget > 0 else 0,
        burn_rate_per_hour=burn_rate,
        hours_remaining=hours_remaining,
        projected_total_usd=projected_total,
        breakdown_by_engine={
            'native': cost_controller.get_spent_by_engine(operator_id, 'native'),
            'acs': cost_controller.get_spent_by_engine(operator_id, 'acs'),
            'tde': cost_controller.get_spent_by_engine(operator_id, 'tde'),
        }
    )
```

```python
@dataclass
class ExecutionContext:
    # ... existing fields ...
    
    # Interrupt support
    paused: bool = False  # If true, turn paused at next checkpoint
    pause_snapshot: Optional[Dict[str, Any]] = None  # Snapshot when paused
    redirect_engine: Optional[str] = None  # If set, redirect to this engine
    
    async def check_interrupt(self) -> None:
        """Called every 10ms during turn execution (at stage boundaries)."""
        if self.paused:
            # Save current state
            self.pause_snapshot = await self.capture_snapshot()
            
            # Wait for resume signal
            while self.paused:
                await asyncio.sleep(0.1)
            
            # Check if redirected
            if self.redirect_engine:
                raise RedirectExecutionException(self.redirect_engine)
            
            # Resume from checkpoint
            await self.restore_snapshot(self.pause_snapshot)
```

```python
@app.post("/api/interrupt/pause")
async def pause_turn(turn_id: str, operator_id: str) -> PauseResponse:
    """Pause current turn execution."""
    context = get_execution_context(turn_id)
    context.paused = True
    
    # Wait for pause to be captured
    while context.pause_snapshot is None:
        await asyncio.sleep(10)  # 10ms check interval
    
    return PauseResponse(
        turn_id=turn_id,
        snapshot=context.pause_snapshot,
        elapsed_time_ms=context.elapsed_time_ms()
    )


@app.post("/api/interrupt/resume")
async def resume_turn(turn_id: str, operator_id: str) -> ResumeResponse:
    """Resume paused turn."""
    context = get_execution_context(turn_id)
    
    if not context.paused:
        raise ValueError("Turn not paused")
    
    context.paused = False
    
    # Wait for turn to complete
    result = await context.wait_for_completion()
    
    return ResumeResponse(
        turn_id=turn_id,
        result=result
    )


@app.post("/api/interrupt/redirect")
async def redirect_turn(turn_id: str, new_engine: str) -> RedirectResponse:
    """Redirect turn to different engine."""
    context = get_execution_context(turn_id)
    
    if new_engine not in ['native', 'acs', 'tde']:
        raise ValueError(f"Invalid engine: {new_engine}")
    
    # Set redirect flag
    context.redirect_engine = new_engine
    
    # Pause first (to capture checkpoint)
    context.paused = True
    while context.pause_snapshot is None:
        await asyncio.sleep(10)
    
    # Resume (will redirect on next checkpoint)
    context.paused = False
    
    # Wait for redirect + completion
    try:
        result = await context.wait_for_completion()
    except RedirectExecutionException:
        # Re-execute with new engine
        result = await execute_with_engine(
            new_engine,
            context.user_input,
            context.execution_context_snapshot
        )
    
    return RedirectResponse(
        turn_id=turn_id,
        original_engine='native',
        new_engine=new_engine,
        result=result
    )
```

```tsx
          className={rating === 1 ? 'selected' : ''}
```
