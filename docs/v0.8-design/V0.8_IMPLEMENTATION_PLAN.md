## Executive Summary

v0.8 enables **offline operation** with local LLM fallback, deterministic operation queue, and safe state reconciliation. Operators can queue tasks while offline (100% reliability, zero data loss), chat with fallback Llama 2 7B (~90% of Claude quality), and seamlessly sync when reconnected. System maintains hash-chain audit integrity, honors GDPR erasure requests even offline, and gracefully degrades features (chat works offline, Vibe Engineering requires streaming).

**Key invariants:**
- SQLite operation queue with deterministic replay (sequence number + hash-chain)
- Last-Write-Wins + custom merge for settings/fingerprint
- Graceful degradation matrix shows operator what works offline
- Sync verification proves offline state matches online (deterministic replay proof)

```python
                        os.path.getmtime(file_path)

                    last_verified_at=now(),
                    audit_hash=await audit_hash_chain(),
                )
        
        # Download
        print(f"Downloading {model_name}...")
        
        async with aiohttp.ClientSession() as session:
            async with session.get(
                model_config["file_url"],
                timeout=aiohttp.ClientTimeout(total=3600),
            ) as resp:
                if resp.status != 200:
                    raise Exception(f"Download failed: {resp.status}")
                
                downloaded = 0
                with open(file_path, "wb") as f:
                    async for chunk in resp.content.iter_chunked(1024*1024):
                        f.write(chunk)
                        downloaded += len(chunk)
                        
                        # Progress
                        pct = (downloaded / (model_config["file_size_gb"] * 1024*1024*1024)) * 100
                        print(f"  {pct:.1f}%", end="\r")
        
        # Verify checksum
        checksum = await self._compute_checksum(file_path)
        if checksum != model_config["checksum"]:
            os.remove(file_path)
            raise Exception(f"Checksum mismatch: {checksum}")
        
        return LocalLLMInfo(
            model_name=model_name,
            model_id="llama-2-7b",
            version="v1.0.0",
            quantization="4-bit",
            file_size_gb=model_config["file_size_gb"],
            file_path=file_path,
            checksum_sha256=checksum,
            context_length=model_config["context"],

            last_verified_at=now(),
            audit_hash=await audit_hash_chain(),

    async def _compute_checksum(self, file_path: str) -> str:
        """Compute SHA256 checksum of file."""
        import hashlib
        
        sha256 = hashlib.sha256()
        with open(file_path, "rb") as f:
            for chunk in iter(lambda: f.read(4096), b""):
                sha256.update(chunk)
        
        return sha256.hexdigest()
```

```python
from dataclasses import dataclass
from typing import Dict, Any, Optional, Literal
from datetime import datetime
import sqlite3
import json

@dataclass(frozen=True)
class QueuedOperation:
    """Immutable queued operation."""
    
    # Identity
    operation_id: str               # UUID
    sequence_number: int            # [1, 2, 3, ...] (deterministic order)
    
    # Operation details
    operator_id: str
    tenant_id: str
    task_type: str                  # "chat", "decision", "analysis"
    
    # Payload
    payload: Dict[str, Any]         # {"prompt": "...", "context": {...}}
    
    # State
    status: Literal[
        "queued",
        "processing",
        "completed",
        "failed",
        "retrying",
    ]
    retry_count: int
    
    # Result (if completed)
    result: Optional[Dict[str, Any]]
    error_message: Optional[str]
    
    # Metadata
    created_at: datetime            # local time
    processed_at: Optional[datetime]
    synced_at: Optional[datetime]   # when synced to server
    
    # Hash-chain integrity
    previous_hash: str              # hash of previous operation
    operation_hash: str             # hash of this operation
    audit_hash: str

class OperationQueueStore:
    """SQLite-backed queue store."""
    
    def __init__(self, db_path: str):
        self.db_path = db_path
        self._ensure_schema()
    
    def _ensure_schema(self):
        """Create SQLite schema if not exists."""
        
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                CREATE TABLE IF NOT EXISTS operation_queue (
                    operation_id TEXT PRIMARY KEY,
                    sequence_number INTEGER UNIQUE NOT NULL,
                    
                    operator_id TEXT NOT NULL,
                    tenant_id TEXT NOT NULL,
                    task_type TEXT NOT NULL,
                    
                    payload TEXT NOT NULL,  -- JSON
                    
                    status TEXT NOT NULL,
                    retry_count INTEGER DEFAULT 0,
                    
                    result TEXT,            -- JSON
                    error_message TEXT,
                    
                    created_at TEXT NOT NULL,
                    processed_at TEXT,
                    synced_at TEXT,
                    
                    previous_hash TEXT,
                    operation_hash TEXT NOT NULL,
                    audit_hash TEXT NOT NULL,
                    
                    INDEX idx_status (status),
                    INDEX idx_tenant (tenant_id),
                    INDEX idx_sequence (sequence_number)
                )
            """)
            
            conn.commit()
    
    async def enqueue(
        self,
        operator_id: str,
        tenant_id: str,
        task_type: str,
        payload: Dict[str, Any],
    ) -> QueuedOperation:
        """
        Add operation to queue.
        
        Algorithm:
        1. Get next sequence number (max + 1)
        2. Compute hash-chain link (hash of previous op)
        3. Compute operation hash
        4. Insert into SQLite
        5. Return QueuedOperation
        """
        
        operation_id = uuid4().hex
        
        with sqlite3.connect(self.db_path) as conn:
            # Get next sequence
            result = conn.execute(
                "SELECT MAX(sequence_number) FROM operation_queue"
            ).fetchone()
            
            sequence_number = (result[0] or 0) + 1
            
            # Get previous hash
            prev_result = conn.execute(
                """
                SELECT operation_hash FROM operation_queue
                WHERE sequence_number = ?
                """,
                (sequence_number - 1,),
            ).fetchone()
            
            previous_hash = prev_result[0] if prev_result else "0x00"
            
            # Compute hashes
            operation_hash = self._compute_hash({
                "operation_id": operation_id,
                "sequence_number": sequence_number,
                "task_type": task_type,
                "payload": payload,
                "previous_hash": previous_hash,
            })
            
            audit_hash = self._compute_hash({
                "operation_hash": operation_hash,
                "timestamp": now().isoformat(),
            })
            
            # Insert
            conn.execute(
                """
                INSERT INTO operation_queue (
                    operation_id, sequence_number, operator_id, tenant_id,
                    task_type, payload, status, created_at,
                    previous_hash, operation_hash, audit_hash
                ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    operation_id,
                    sequence_number,
                    operator_id,
                    tenant_id,
                    task_type,
                    json.dumps(payload),
                    "queued",
                    now().isoformat(),
                    previous_hash,
                    operation_hash,
                    audit_hash,
                ),
            )
            
            conn.commit()
            
            return QueuedOperation(
                operation_id=operation_id,
                sequence_number=sequence_number,
                operator_id=operator_id,
                tenant_id=tenant_id,
                task_type=task_type,
                payload=payload,
                status="queued",
                retry_count=0,
                result=None,
                error_message=None,
                created_at=now(),
                processed_at=None,
                synced_at=None,
                previous_hash=previous_hash,
                operation_hash=operation_hash,
                audit_hash=audit_hash,
            )
    
    async def process_queue(
        self,
        tenant_id: str,
    ) -> List[QueuedOperation]:
        """
        Process all queued operations in order.
        
        Returns list of processed operations.
        """
        
        with sqlite3.connect(self.db_path) as conn:
            # Get queued operations in sequence order
            rows = conn.execute(
                """
                SELECT * FROM operation_queue
                WHERE tenant_id = ? AND status IN ('queued', 'retrying')
                ORDER BY sequence_number ASC
                """,
                (tenant_id,),
            ).fetchall()
        
        processed = []
        for row in rows:
            op = self._row_to_operation(row)
            
            # Execute operation
            try:
                result = await self._execute_operation(op)
                
                # Update status to completed
                with sqlite3.connect(self.db_path) as conn:
                    conn.execute(
                        """
                        UPDATE operation_queue
                        SET status = ?, result = ?, processed_at = ?, synced_at = ?
                        WHERE operation_id = ?
                        """,
                        (
                            "completed",
                            json.dumps(result),
                            now().isoformat(),
                            now().isoformat(),
                            op.operation_id,
                        ),
                    )
                    conn.commit()
                
                processed.append(op)
            
            except Exception as e:
                # Update status to failed/retrying
                with sqlite3.connect(self.db_path) as conn:
                    conn.execute(
                        """
                        UPDATE operation_queue
                        SET status = ?, error_message = ?, retry_count = retry_count + 1
                        WHERE operation_id = ?
                        """,
                        (
                            "retrying" if op.retry_count < 3 else "failed",
                            str(e),
                            op.operation_id,
                        ),
                    )
                    conn.commit()
        
        return processed
    
    def _compute_hash(self, data: Dict[str, Any]) -> str:
        """Compute SHA256 hash of data."""
        import hashlib
        
        payload_str = json.dumps(data, sort_keys=True)
        return hashlib.sha256(payload_str.encode()).hexdigest()
```

#### 2.2 Deterministic Replay

**File:** `core/offline/deterministic_replay.py` (NEW)

```python
class DeterministicReplay:
    """Replay queued operations with deterministic results."""
    
    async def replay_operations(
        self,
        operations: List[QueuedOperation],
        operator_id: str,
        tenant_id: str,
    ) -> List[Dict[str, Any]]:
        """
        Replay operations deterministically.
        
        Algorithm:
        1. For each operation in sequence order:
           a. Restore execution context (fingerprint, settings)
           b. Mock all external calls (APIs, databases)
           c. Execute operation with mocked state
           d. Capture result
        2. Return list of results (in sequence order)
        3. Later: compare against stored results for verification
        """
        
        results = []
        context_snapshot = None
        
        for op in operations:
            # Restore context (fingerprint, settings at time of operation)
            context_snapshot = await self._restore_context(
                operator_id,
                tenant_id,
                op.created_at,  # restore state as of this time
            )
            
            # Mock all external calls
            mock_manager = MockManager(context_snapshot)
            
            # Execute deterministically
            result = await mock_manager.execute_operation(op)
            
            results.append({
                "operation_id": op.operation_id,
                "sequence_number": op.sequence_number,
                "result": result,
            })
        
        return results
    
    async def verify_replay(
        self,
        stored_operations: List[QueuedOperation],
        replayed_results: List[Dict[str, Any]],
    ) -> Tuple[bool, List[str]]:
        """
        Verify replay matches stored results.
        
        Returns:
            (all_match, mismatches)
        """
        
        mismatches = []
        
        for stored, replayed in zip(stored_operations, replayed_results):
            stored_result = json.loads(stored.result) if stored.result else {}
            replayed_result = replayed["result"]
            
            if stored_result != replayed_result:
                mismatches.append(
                    f"Op {stored.sequence_number}: "
                    f"stored={stored_result}, "
                    f"replayed={replayed_result}"
                )
        
        return (len(mismatches) == 0, mismatches)
```

```python
class HashChainVerifier:
    """Verify operation queue hash-chain integrity."""
    
    async def verify_hash_chain(
        self,
        operations: List[QueuedOperation],
    ) -> Tuple[bool, List[str]]:
        """
        Verify every operation links to previous.
        
        Algorithm:
        1. For each operation i:
           a. Recompute hash of operation i-1
           b. Compare to operation[i].previous_hash
           c. Flag mismatch
        2. Return (all_valid, list_of_errors)
        """
        
        errors = []
        
        for i, op in enumerate(operations):
            if i == 0:
                # First operation: previous_hash should be "0x00"
                if op.previous_hash != "0x00":
                    errors.append(
                        f"Op {i}: first operation should have previous_hash='0x00', "
                        f"got '{op.previous_hash}'"
                    )
            else:
                prev_op = operations[i - 1]
                
                # Recompute prev_op's hash
                recomputed_hash = self._compute_operation_hash(prev_op)
                
                if recomputed_hash != op.previous_hash:
                    errors.append(
                        f"Op {i}: hash-chain broken, "
                        f"expected previous_hash={recomputed_hash}, "
                        f"got {op.previous_hash}"
                    )
        
        return (len(errors) == 0, errors)
    
    def _compute_operation_hash(self, op: QueuedOperation) -> str:
        """Recompute hash of operation."""
        import hashlib
        
        payload = {
            "operation_id": op.operation_id,
            "sequence_number": op.sequence_number,
            "task_type": op.task_type,
            "payload": op.payload,
            "previous_hash": op.previous_hash,
        }
        
        payload_str = json.dumps(payload, sort_keys=True)
        return hashlib.sha256(payload_str.encode()).hexdigest()
```

```python
        Merge local (offline) state with remote (server) state.
        
        Algorithm:
        1. For each field in local_state:
           a. If field not in remote_state: use local
           b. If field is timestamp-tagged: LWW (later timestamp wins)
           c. If field is array (plugins, flags): merge arrays
           d. If field is object (fingerprint): recursive merge
        2. Return merged state + list of conflicts
        
        Example:
        local:  { settings: { plugins_enabled: [p1, p2], lang: "de" }, fingerprint: {...} }
        remote: { settings: { plugins_enabled: [p3], lang: "en" }, fingerprint: {...} }
        merged: { settings: { plugins_enabled: [p1, p2, p3], lang: "en" }, fingerprint: merged }
        """
        
        merged = {}
        conflicts = []
        
        # Merge settings
        local_settings = local_state.get("settings", {})
        remote_settings = remote_state.get("settings", {})
        
        merged["settings"] = await self._merge_settings(
            local_settings,
            remote_settings,
            operator_id,
        )
        
        # Merge fingerprint (if present)
        local_fp = local_state.get("fingerprint")
        remote_fp = remote_state.get("fingerprint")
        
        if local_fp and remote_fp:
            merged_fp, fp_conflicts = await self._merge_fingerprint(
                local_fp,
                remote_fp,
            )
            merged["fingerprint"] = merged_fp
            conflicts.extend(fp_conflicts)
        else:
            merged["fingerprint"] = local_fp or remote_fp
        
        return (merged, conflicts)
    
    async def _merge_settings(
        self,
        local: Dict[str, Any],
        remote: Dict[str, Any],
        operator_id: str,
    ) -> Dict[str, Any]:
        """
        Merge settings using custom rules.
        
        Rules:
        - language: LWW (last write wins)
        - plugins_enabled: array merge (union of local + remote)
        - feature_flags: LWW per flag
        - privacy_consent: LWW (later timestamp)
        """
        
        merged = {}
        
        # Simple fields: LWW
        for field in ["language", "theme"]:
            if field in local or field in remote:
                local_val = local.get(field)
                remote_val = remote.get(field)
                
                # Use local if both set (offline happened later)
                merged[field] = local_val if local_val else remote_val
        
        # Array merge: plugins_enabled
        local_plugins = set(local.get("plugins_enabled", []))
        remote_plugins = set(remote.get("plugins_enabled", []))
        
        merged["plugins_enabled"] = list(local_plugins | remote_plugins)
        
        # Feature flags: LWW per flag
        merged["feature_flags"] = {}
        
        all_flags = set(
            local.get("feature_flags", {}).keys() |
            remote.get("feature_flags", {}).keys()
        )
        
        for flag in all_flags:
            local_flag_val = local.get("feature_flags", {}).get(flag)
            remote_flag_val = remote.get("feature_flags", {}).get(flag)
            
            merged["feature_flags"][flag] = (
                local_flag_val if local_flag_val is not None else remote_flag_val
            )
        
        return merged
    
    async def _merge_fingerprint(
        self,
        local: Dict[str, Any],
        remote: Dict[str, Any],
    ) -> Tuple[Dict[str, Any], List[str]]:
        """
        Merge operator fingerprints.
        
        Rules:
        - Use whichever has higher sample_count (more data)
        - If equal, use local (offline)
        """
        
        conflicts = []
        
        local_samples = local.get("sample_count", 0)
        remote_samples = remote.get("sample_count", 0)
        
        if local_samples >= remote_samples:
            merged = local
        else:
            merged = remote
            
            if local_samples > 0:
                conflicts.append(
                    f"Fingerprint: remote has more data "
                    f"(local={local_samples}, remote={remote_samples}), "
                    f"using remote"
                )
        
        return (merged, conflicts)
```

```python
@dataclass(frozen=True)
class SyncVerificationReport:
    """Immutable sync verification report."""
    
    # Identity
    operator_id: str
    tenant_id: str
    sync_id: str
    
    # Verification results
    hash_chain_valid: bool
    replay_matches_stored: bool
    conflicts_detected: List[str]
    
    # Metrics
    operations_verified: int
    operations_failed: int
    replay_time_ms: float
    
    # Status
    status: Literal[
        "verified",
        "chain_broken",
        "replay_mismatch",
        "conflicts_unresolved",
    ]
    
    # Audit
    verified_at: datetime
    audit_hash: str

class SyncVerifier:
    """Verify offline state matches online after sync."""
    
    async def verify_sync(
        self,
        operator_id: str,
        tenant_id: str,
    ) -> SyncVerificationReport:
        """
        Verify offline state by replaying operations.
        
        Algorithm:
        1. Load offline operation queue
        2. Verify hash-chain integrity (every op links to previous)
        3. Replay all operations deterministically
        4. Compare replayed results to stored results
        5. Return verification report
        
        If any step fails, halt and report error.
        """
        
        # Step 1: Load queue
        queue_store = OperationQueueStore("~/.corvin/operation_queue.db")
        operations = await queue_store.load_all(tenant_id)
        
        # Step 2: Verify hash-chain
        start = time.time()
        
        verifier = HashChainVerifier()
        chain_valid, chain_errors = await verifier.verify_hash_chain(operations)
        
        if not chain_valid:
            return SyncVerificationReport(
                operator_id=operator_id,
                tenant_id=tenant_id,
                sync_id=uuid4().hex,
                hash_chain_valid=False,
                replay_matches_stored=False,
                conflicts_detected=[],
                operations_verified=0,
                operations_failed=len(chain_errors),
                replay_time_ms=(time.time() - start) * 1000,
                status="chain_broken",
                verified_at=now(),
                audit_hash=await audit_hash_chain(),
            )
        
        # Step 3: Replay deterministically
        replay_engine = DeterministicReplay()
        replayed_results = await replay_engine.replay_operations(
            operations,
            operator_id,
            tenant_id,
        )
        
        # Step 4: Compare to stored
        all_match, mismatches = await replay_engine.verify_replay(
            operations,
            replayed_results,
        )
        
        elapsed = time.time() - start
        
        if not all_match:
            return SyncVerificationReport(
                operator_id=operator_id,
                tenant_id=tenant_id,
                sync_id=uuid4().hex,
                hash_chain_valid=True,
                replay_matches_stored=False,
                conflicts_detected=mismatches,
                operations_verified=len(operations) - len(mismatches),
                operations_failed=len(mismatches),
                replay_time_ms=elapsed * 1000,
                status="replay_mismatch",
                verified_at=now(),
                audit_hash=await audit_hash_chain(),
            )
        
        # All verified!
        return SyncVerificationReport(
            operator_id=operator_id,
            tenant_id=tenant_id,
            sync_id=uuid4().hex,
            hash_chain_valid=True,
            replay_matches_stored=True,
            conflicts_detected=[],
            operations_verified=len(operations),
            operations_failed=0,
            replay_time_ms=elapsed * 1000,
            status="verified",
            verified_at=now(),
            audit_hash=await audit_hash_chain(),
        )
```

**File:** `core/offline/sync_completion.py` (NEW)

```python
async def complete_sync(
    operator_id: str,
    tenant_id: str,
) -> dict:
    """
    Complete sync: verify, merge states, process queue.
    
    Algorithm:
    1. Verify offline state (hash-chain + replay)
    2. Merge operator state (CRDT merge)
    3. Process operation queue
    4. Clean up offline data (operation queue)
    5. Return summary
    """
    
    # Step 1: Verify
    verifier = SyncVerifier()
    verification = await verifier.verify_sync(operator_id, tenant_id)
    
    if verification.status != "verified":
        return {
            "status": "verification_failed",
            "reason": verification.status,
            "details": verification.conflicts_detected,
        }
    
    # Step 2: Merge states
    local_state = await load_operator_state(operator_id, tenant_id, source="local")
    remote_state = await load_operator_state(operator_id, tenant_id, source="remote")
    
    merger = OperatorStateMerger()
    merged_state, conflicts = await merger.merge_states(
        local_state,
        remote_state,
        operator_id,
        config=CRDTMergeConfig(strategy="lww", timestamp_resolution_ms=1000),
    )
    
    if conflicts:
        # Return conflicts for operator to resolve
        return {
            "status": "conflicts",
            "conflicts": conflicts,
        }
    
    # Step 3: Process queue
    queue_store = OperationQueueStore("~/.corvin/operation_queue.db")
    processed = await queue_store.process_queue(tenant_id)
    
    # Step 4: Clean up
    await queue_store.clear_processed(tenant_id)
    
    # Step 5: Summary
    return {
        "status": "completed",
        "operations_synced": len(processed),
        "verification": {
            "hash_chain_valid": verification.hash_chain_valid,
            "replay_time_ms": verification.replay_time_ms,
        },
        "state_merged": True,
        "summary": f"Synced {len(processed)} operations, merged state, no conflicts",
    }
```
